cmake_minimum_required(VERSION 3.16)
project(CircuitSolver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(circuit STATIC circuit.cpp)
target_include_directories(circuit PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(circuit_solver main.cpp)
target_link_libraries(circuit_solver PRIVATE circuit)
enable_testing()
add_executable(circuit_tests tests.cpp)
target_link_libraries(circuit_tests PRIVATE circuit)
add_test(NAME circuit_tests COMMAND circuit_tests)
